  trainingset<-data[training,]
  testingset<-data[-training,]
  ###fill in the codes to get the predictions errors
  ### update the totalCVerror
  model<-lm(y~x, data=trainingset)
  prediction<-predict.lm(model, data=testingset)  #check the syntax of predict.lm
  errors<-sum((testingset$y-prediction)^2)
  totalCVerror<-totalCVerror+errors
}
